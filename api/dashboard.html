from fastapi import FastAPI, UploadFile from classifier import WasteClassifier import asyncio app = FastAPI() clf = WasteClassifier("models/resnet50_waste.pt") @app.post("/classify") async def classify_waste(file: UploadFile): img_bytes = await file.read() result = await asyncio.to_thread(clf.predict, img_bytes) if result["confidence"] < 0.70: await flag_for_retraining(img_bytes, result) return result # {class, confidence, bbox} # Run: uvicorn main:app --workers 4 --host 0.0.0.0 --port 8000
